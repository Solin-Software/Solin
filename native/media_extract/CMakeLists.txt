cmake_minimum_required(VERSION 3.25)
project(solin_media_extract VERSION 1.0.0 LANGUAGES CXX)

set(SOLIN_FFMPEG_ROOT "" CACHE PATH "Packaged FFmpeg SDK root (include/libavformat/avformat.h)")
set(SOLIN_FFMPEG_RUNTIME_SUBDIR "." CACHE STRING "FFmpeg runtime directory relative to the executable")
option(SOLIN_MEDIA_EXTRACT_BUILD_TESTS "Build native protocol and geometry tests" ON)
if(SOLIN_FFMPEG_ROOT)
    set(SOLIN_FFMPEG_INCLUDE_DIR "${SOLIN_FFMPEG_ROOT}/include")
    if(CMAKE_LIBRARY_ARCHITECTURE AND EXISTS
       "${SOLIN_FFMPEG_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}/libavformat/avformat.h")
        string(APPEND SOLIN_FFMPEG_INCLUDE_DIR "/${CMAKE_LIBRARY_ARCHITECTURE}")
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SOLIN_FFMPEG_SDK REQUIRED
        libavformat>=60.16.100 libavcodec>=60.31.102 libavutil>=58.29.100 libswscale>=7.5.100)
    find_path(SOLIN_FFMPEG_INCLUDE_DIR libavformat/avformat.h
        HINTS ${SOLIN_FFMPEG_SDK_INCLUDE_DIRS} NO_CACHE REQUIRED)
else()
    message(FATAL_ERROR "SOLIN_FFMPEG_ROOT must point to the builder-provided FFmpeg SDK")
endif()
if(NOT EXISTS "${SOLIN_FFMPEG_INCLUDE_DIR}/libavformat/avformat.h")
    message(FATAL_ERROR "SOLIN_FFMPEG_ROOT is missing include/libavformat/avformat.h")
endif()
string(REPLACE "\\" "/" SOLIN_FFMPEG_RUNTIME_SUBDIR "${SOLIN_FFMPEG_RUNTIME_SUBDIR}")
if(IS_ABSOLUTE "${SOLIN_FFMPEG_RUNTIME_SUBDIR}" OR SOLIN_FFMPEG_RUNTIME_SUBDIR MATCHES "(^|/)\\.\\.(/|$)")
    message(FATAL_ERROR "SOLIN_FFMPEG_RUNTIME_SUBDIR must stay below the executable directory")
endif()

function(solin_media_target target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

add_library(solin_media_contract STATIC src/protocol.cpp src/geometry.cpp)
target_include_directories(solin_media_contract PUBLIC src)
solin_media_target(solin_media_contract)
add_executable(solin-media-extract src/main.cpp src/ffmpeg_api.cpp src/extractor.cpp)
target_include_directories(solin-media-extract SYSTEM PRIVATE "${SOLIN_FFMPEG_INCLUDE_DIR}")
target_compile_definitions(solin-media-extract PRIVATE
    SOLIN_FFMPEG_RUNTIME_SUBDIR="${SOLIN_FFMPEG_RUNTIME_SUBDIR}")
target_link_libraries(solin-media-extract PRIVATE solin_media_contract ${CMAKE_DL_LIBS})
solin_media_target(solin-media-extract)
if(WIN32)
    target_compile_definitions(solin-media-extract PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MINGW)
        target_link_options(solin-media-extract PRIVATE -municode)
    endif()
endif()
install(TARGETS solin-media-extract RUNTIME DESTINATION .)

if(SOLIN_MEDIA_EXTRACT_BUILD_TESTS)
    enable_testing()
    foreach(component protocol geometry deadline)
        add_executable(solin-media-${component}-tests tests/${component}_tests.cpp)
        target_link_libraries(solin-media-${component}-tests PRIVATE solin_media_contract)
        solin_media_target(solin-media-${component}-tests)
        add_test(NAME media_${component} COMMAND solin-media-${component}-tests)
    endforeach()
endif()
